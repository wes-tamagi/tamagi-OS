# Pacotes RPM adicionados à base ghcr.io/ublue-os/base-main (um por linha;

# pre-configured (/etc/chromium/tamagi.conf). Other browsers are optional via `tamagi browser`.

# ── Terminal e desenvolvimento (estilo Omarchy) ──
# foot: terminal padrão (Super+Enter e apps de terminal): leve e rápido para
# abrir. O cosmic-term continua instalado e no lançador.
foot
# Ptyxis: o terminal do sistema (Super+Enter). Conhece as caixas distrobox e
# os contêineres do podman: abre uma aba direto dentro deles pelo menu.
ptyxis
git
btop
fastfetch
gum
# Fora da imagem, na caixa de desenvolvimento (`tamagi devbox`), que as exporta
# para o sistema ao ser criada: neovim, lazygit, gh, uv, glow, hyperfine,
# direnv, yq e ShellCheck (ver /usr/libexec/tamagi/devbox-provision).
# Dev Containers com compose (o VS Code é opcional: tamagi install vscode).
podman-compose

# ── Desempenho ──
# Jogos (Steam, steam-devices, gamescope, MangoHud, GameMode, Lutris, Wine)
# não vêm na imagem: entram nativamente, sob demanda, numa camada reversível
# (`tamagi gaming native`), como os pacotes que o Bazzite traz embutidos.
# Escalonador de CPU em espaço de usuário via BPF (sched_ext), gerido por tamagi-perf.
scx_rusty

# ── Camadas sobre a imagem (tamagi-layer: systemd-sysext em erofs) ──
erofs-utils
# Carregador de boot: systemd-boot, como no openSUSE Aeon (o bootc o instala
# com o backend composefs; o kernel fica na partição EFI e não há /boot).
systemd-boot-unsigned

# ── Boot que se conserta: verificação de saúde e rollback automático ──
# Histórico da pasta pessoal (snapshots btrfs a cada hora): tamagi-snap.
snapper
# Manutenção do btrfs como no openSUSE MicroOS/Aeon: scrub mensal (acha
# corrupção silenciosa) e balance semanal, em timers do systemd.
btrfsmaintenance

# ── Segurança (inspirado no GrapheneOS; veja tamagi-harden) ──
usbguard
# hardened_malloc: COPR do secureblue; instalado mas só ativado no modo estrito.
hardened_malloc
